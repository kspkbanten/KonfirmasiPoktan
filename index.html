function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('Konfirmasi');
  if (!sh) {
    sh = ss.insertSheet('Konfirmasi');
    sh.appendRow(['updated_at','type','id_poktan','nama_kab','nama_kec','nama_desa','nama_poktan','konfirmasi','alasan','nama_pic','jabatan']);
  }
  var data = sh.getDataRange().getValues();
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var o = {};
    for (var j = 0; j < headers.length; j++) o[headers[j]] = data[i][j];
    if (o.id_poktan) rows.push(o);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true, rows:rows}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('Konfirmasi');
  if (!sh) {
    sh = ss.insertSheet('Konfirmasi');
    sh.appendRow(['updated_at','type','id_poktan','nama_kab','nama_kec','nama_desa','nama_poktan','konfirmasi','alasan','nama_pic','jabatan']);
  }
  var body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:String(err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var headers = sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0];
  var idCol = headers.indexOf('id_poktan') + 1;
  var typeCol = headers.indexOf('type') + 1;
  var last = sh.getLastRow();
  var found = 0;
  if (last >= 2) {
    var ids = sh.getRange(2, idCol, last, idCol).getValues();
    var types = sh.getRange(2, typeCol, last, typeCol).getValues();
    for (var i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(body.id_poktan) && String(types[i][0]) === String(body.type)) {
        found = i + 2; break;
      }
    }
  }
  var row = [
    body.updated_at || new Date().toISOString(),
    body.type || '',
    body.id_poktan || '',
    body.nama_kab || '',
    body.nama_kec || '',
    body.nama_desa || '',
    body.nama_poktan || '',
    body.konfirmasi || '',
    body.alasan || '',
    body.nama_pic || '',
    body.jabatan || ''
  ];
  if (found) sh.getRange(found, 1, found, row.length).setValues([row]);
  else sh.appendRow(row);
  return ContentService.createTextOutput(JSON.stringify({ok:true, upsert: found ? 'update' : 'insert'}))
    .setMimeType(ContentService.MimeType.JSON);
}

✓ Pertahankan/Kelompok Ideal

Sesuai Kerangka Konsep Keaktifan Poktan (Pusat Data dan Teknologi Informasi, Kemendukbangga/BKKBN). Poktan dipertahankan apabila memenuhi seluruh unsur berikut:

Jumlah anggota BKB, BKR, BKL 20–50 keluarga
UPPKA 10–50 keluarga

BKB, BKR, BKL, UPPKA Kehadiran anggota minimal 50% setiap bulan

◐ Perbaikan

Memenuhi unsur kader, kehadiran, dan pelaporan, tetapi jumlah anggota di luar rentang ideal (terlalu besar). Perlu penyesuaian struktur/pembagian kelompok:

UnsurJenis PoktanKondisi Perbaikan
Kader / konselor Semua Sudah memenuhi minimal (≥2)
Jumlah anggota BKB, BKR, BKL, UPPKA Lebih dari 50 keluarga
PIK-R Pengurus/pendidik/konselor sebaya < 20 orang (jika masih memenuhi syarat lain sebagian)
Kehadiran & pelaporan BKB, BKR, BKL, UPPKA Kehadiran ≥50% dan pelaporan rutin masih terpenuhi
